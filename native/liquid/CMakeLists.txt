# Android JNI library that statically links a pinned llama.cpp runtime.
cmake_minimum_required(VERSION 3.22)
project(vmr_liquid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
if(NOT DEFINED LLAMA_SOURCE OR NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Specify -DLLAMA_SOURCE pinned checkout")
endif()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_BINARY_DIR}/pinned-llama" EXCLUDE_FROM_ALL)
add_library(vmr_liquid SHARED vmr_liquid.cpp)
target_include_directories(vmr_liquid PRIVATE "${LLAMA_SOURCE}/include")
target_link_libraries(vmr_liquid PRIVATE llama android log)
target_compile_options(vmr_liquid PRIVATE -fvisibility=hidden -fexceptions)
target_link_options(vmr_liquid PRIVATE -Wl,--exclude-libs,ALL)
